cmake_minimum_required(VERSION 3.16)
file(READ "${CMAKE_CURRENT_LIST_DIR}/main/version.json" ems_release_json)
string(JSON ems_version GET "${ems_release_json}" version)
set(PROJECT_VER "${ems_version}")
if(NOT DEFINED BUILD_TARGET)
    set(BUILD_TARGET "esp32s3_n16r8")
endif()
if(BUILD_TARGET STREQUAL "atoms3_lite_8mb")
    set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;sdkconfig.defaults.atoms3_lite_8mb")
elseif(BUILD_TARGET STREQUAL "esp32s3_n16r8")
    set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;sdkconfig.defaults.n16r8")
else()
    message(FATAL_ERROR "Unknown BUILD_TARGET: ${BUILD_TARGET}")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
if(TEE_HTTP_CONTROL_TRIAL)
    list(APPEND SDKCONFIG_DEFAULTS "sdkconfig.defaults.http_control_trial")
endif()
message(STATUS "TeeNet hardware target: ${BUILD_TARGET}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(wallbox_ems)
